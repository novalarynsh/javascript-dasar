function sapa() {
  // Ambil nilai input dari DOM
  const nama = document.getElementById('inp1').value;
  const output = document.getElementById('out1');

  // Event handling + kondisi if/else
  if (nama.trim() === '') {
    output.innerText = 'Masukkan nama dulu!';
  } else {
    output.innerText = 'Halo, ' + nama + '! Selamat belajar!';
  }
}

// Variabel global yang menyimpan state
let angka = 0;

function ubahCounter(delta) {
  // Logika update nilai
  if (delta === 0) { angka = 0; }
  else { angka += delta; }

  // Manipulasi DOM: tampilkan nilai terbaru
  document.getElementById('counter').innerText = angka;
}

    function tambahTodo() {
  const teks = document.getElementById('todo-inp').value.trim();
  if (!teks) return;

  // Buat elemen baru secara dinamis
  const li = document.createElement('li');
  const span = document.createElement('span');
  span.innerText = teks;

  const tombolHapus = document.createElement('button');
  tombolHapus.innerText = '✕';
  tombolHapus.onclick = function() {
    // Hapus elemen dari DOM
    document.getElementById('todo-list').removeChild(li);
  };

  // Sisipkan ke dalam DOM
  li.appendChild(span);
  li.appendChild(tombolHapus);
  document.getElementById('todo-list').appendChild(li);
}
